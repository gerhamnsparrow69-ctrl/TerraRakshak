// TIER 2: ARDUINO NANO - 6x MQ GAS SENSOR ARRAY 

const int analogPins[6] = {A0, A1, A2, A3, A6, A7};
const int digitalPins[6] = {2, 3, 4, 5, 6, 7};

unsigned long last_send = 0;

void setup() {
  Serial.begin(115200);
  for (int i = 0; i < 6; i++) {
    pinMode(digitalPins[i], INPUT);
  }
}

void loop() {
  // Send telemetry every 100ms (10Hz)
  if (millis() - last_send >= 100) {
    last_send = millis();

    Serial.print("{\"T\":2,\"mq_a\":[");
    for (int i = 0; i < 6; i++) {
      Serial.print(analogRead(analogPins[i]));
      if (i < 5) Serial.print(",");
    }
    Serial.print("],\"mq_d\":[");
    for (int i = 0; i < 6; i++) {
      Serial.print(digitalRead(digitalPins[i]));
      if (i < 5) Serial.print(",");
    }
    Serial.println("]}");
  }
}
