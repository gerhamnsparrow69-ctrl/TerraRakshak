//TIER 1: ARDUINO NANO - DRIVETRAIN, ENCODERS & MOTOR CONTROL

#define LEFT_RPWM 5
#define LEFT_LPWM 6
#define RIGHT_RPWM 9
#define RIGHT_LPWM 10

#define LEFT_EN 7    // ties R_EN+L_EN on the left BTS7960 — see note below
#define RIGHT_EN 8   // same for the right BTS7960

#define ENC_L 2
#define ENC_R 3

volatile long count_L = 0;
volatile long count_R = 0;

unsigned long last_telemetry = 0;
const unsigned long TELEMETRY_INTERVAL_MS = 100;

int target_L = 0;   // -255..255, current commanded PWM, left side
int target_R = 0;   // -255..255, current commanded PWM, right side

unsigned long last_command_time = 0;
const unsigned long COMMAND_TIMEOUT_MS = 500; 

String inputBuffer = "";

void isr_L() { count_L++; }
void isr_R() { count_R++; }

long readCount(volatile long &counter) {
  noInterrupts();
  long value = counter;
  interrupts();
  return value;
}

void driveMotor(int pwmVal, int rpwmPin, int lpwmPin) {
  pwmVal = constrain(pwmVal, -255, 255);
  if (pwmVal >= 0) {
    analogWrite(rpwmPin, pwmVal);
    analogWrite(lpwmPin, 0);
  } else {
    analogWrite(rpwmPin, 0);
    analogWrite(lpwmPin, -pwmVal);
  }
}


void parseCommand(String line) {
  int lIdx = line.indexOf("L:");
  int rIdx = line.indexOf("R:");
  if (lIdx == -1 || rIdx == -1) return; // malformed, ignore

  int lCommaIdx = line.indexOf(',', lIdx);
  String lStr = line.substring(lIdx + 2, lCommaIdx);
  String rStr = line.substring(rIdx + 2);

  target_L = constrain(lStr.toInt(), -255, 255);
  target_R = constrain(rStr.toInt(), -255, 255);
  last_command_time = millis();
}

void setup() {
  Serial.begin(115200);

  pinMode(LEFT_RPWM, OUTPUT);  pinMode(LEFT_LPWM, OUTPUT);
  pinMode(RIGHT_RPWM, OUTPUT); pinMode(RIGHT_LPWM, OUTPUT);
  pinMode(LEFT_EN, OUTPUT);    pinMode(RIGHT_EN, OUTPUT);
  digitalWrite(LEFT_EN, HIGH);
  digitalWrite(RIGHT_EN, HIGH);

  pinMode(ENC_L, INPUT_PULLUP);
  pinMode(ENC_R, INPUT_PULLUP);

  attachInterrupt(digitalPinToInterrupt(ENC_L), isr_L, RISING);
  attachInterrupt(digitalPinToInterrupt(ENC_R), isr_R, RISING);

  last_command_time = millis(); // don't instantly trigger the timeout on boot
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') {
      parseCommand(inputBuffer);
      inputBuffer = "";
    } else {
      inputBuffer += c;
    }
  }


  if (millis() - last_command_time > COMMAND_TIMEOUT_MS) {
    target_L = 0;
    target_R = 0;
  }


  driveMotor(target_L, LEFT_RPWM, LEFT_LPWM);
  driveMotor(target_R, RIGHT_RPWM, RIGHT_LPWM);

  if (millis() - last_telemetry >= TELEMETRY_INTERVAL_MS) {
    last_telemetry = millis();
    long l = readCount(count_L);
    long r = readCount(count_R);

    Serial.print("{\"T\":1,\"left\":");
    Serial.print(l);
    Serial.print(",\"right\":");
    Serial.print(r);
    Serial.println("}");
  }
}
